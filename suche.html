// ------------------------------------------------------------- // Suchindex: wird von den HTML-Seiten gefüllt // ------------------------------------------------------------- const searchIndex = []; // ------------------------------------------------------------- // Synonyme (werden automatisch ergänzt) // ------------------------------------------------------------- const synonyms = { "potenz": ["exponent", "hochzahl", "quadratzahl"], "gleichung": ["term", "variable", "auflösen"], "winkel": ["grad", "geometrie", "dreieck"], "fläche": ["quadratmeter", "rechteck"], "volumen": ["liter", "zylinder"], "funktion": ["graph", "gerade", "parabel"], "proportional": ["verhältnis", "faktor", "steigung"], "bruch": ["zähler", "nenner", "bruchzahl"], "prozent": ["prozente", "rabatt", "anteil"] }; // ------------------------------------------------------------- // Levenshtein-Distanz (Fuzzy-Matching) // ------------------------------------------------------------- function levenshtein(a, b) { const matrix = Array.from({ length: b.length + 1 }, (_, i) => Array.from({ length: a.length + 1 }, (_, j) => i === 0 ? j : j === 0 ? i : 0 ) ); for (let i = 1; i <= b.length; i++) { for (let j = 1; j <= a.length; j++) { matrix[i][j] = Math.min( matrix[i - 1][j] + 1, matrix[i][j - 1] + 1, matrix[i - 1][j - 1] + (a[j - 1] === b[i - 1] ? 0 : 1) ); } } return matrix[b.length][a.length]; } // ------------------------------------------------------------- // Automatische Keyword-Erweiterung // ------------------------------------------------------------- function generateKeywordVariants(baseWords) { const variants = new Set(); baseWords.forEach(word => { const w = word.toLowerCase(); variants.add(w); // Pluralformen variants.add(w + "e"); variants.add(w + "en"); variants.add(w + "er"); variants.add(w + "s"); // Bindestrichvarianten variants.add(w + "-"); variants.add(w + "-formel"); variants.add(w + "-gesetz"); // Kurzformen if (w.length > 4) { variants.add(w.slice(0, 3)); variants.add(w.slice(0, w.length - 2)); } // Tippfehler (Buchstaben vertauschen) for (let i = 0; i < w.length - 1; i++) { variants.add( w.slice(0, i) + w[i + 1] + w[i] + w.slice(i + 2) ); } // Englische Varianten if (w === "potenz") { variants.add("power"); variants.add("power function"); } if (w === "funktion") { variants.add("function"); variants.add("graph"); } if (w === "winkel") { variants.add("angle"); } if (w === "fläche") { variants.add("area"); } if (w === "volumen") { variants.add("volume"); } }); return Array.from(variants); } // ------------------------------------------------------------- // Haupt-Suchfunktion // ------------------------------------------------------------- function search(term) { term = term.toLowerCase().trim(); if (!term) return []; let terms = term.split(/\s+/); // Synonyme ergänzen let expandedTerms = new Set(terms); terms.forEach(t => { if (synonyms[t]) { synonyms[t].forEach(s => expandedTerms.add(s)); } }); let results = []; searchIndex.forEach(entry => { let score = 0; expandedTerms.forEach(t => { entry.keywords.forEach(k => { if (k.includes(t)) score += 3; if (levenshtein(k, t) <= 2) score += 2; }); }); if (score > 0) { results.push({ entry, score }); } }); results.sort((a, b) => b.score - a.score); return results.map(r => r.entry); } // ------------------------------------------------------------- // Navigation-Suche aktivieren // ------------------------------------------------------------- document.addEventListener("DOMContentLoaded", () => { const navInput = document.getElementById("navSearch"); const navResults = document.getElementById("navResults"); if (navInput && navResults) { navInput.addEventListener("input", () => { const term = navInput.value.trim(); const results = search(term); navResults.innerHTML = ""; results.slice(0, 5).forEach(r => { const div = document.createElement("div"); div.innerHTML = `${r.title}`; navResults.appendChild(div); }); }); } }); // ------------------------------------------------------------- // Quick-Suche aktivieren // ------------------------------------------------------------- document.addEventListener("DOMContentLoaded", () => { const quickInput = document.getElementById("quickSearch"); const quickResults = document.getElementById("quickResults"); if (quickInput && quickResults) { quickInput.addEventListener("input", () => { const term = quickInput.value.trim(); const results = search(term); quickResults.innerHTML = ""; results.forEach(r => { const div = document.createElement("div"); div.innerHTML = `${r.title}`; quickResults.appendChild(div); }); }); } });
