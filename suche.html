// ------------------------------------------------------------- // Suchindex: wird von den HTML-Seiten gefüllt // ------------------------------------------------------------- const searchIndex = []; // ------------------------------------------------------------- // Synonyme (werden automatisch ergänzt) // ------------------------------------------------------------- const synonyms = { "potenz": ["exponent", "hochzahl", "zahl hoch", "quadratzahl"], "gleichung": ["term", "variable", "auflösen"], "winkel": ["grad", "geometrie", "dreieck"], "fläche": ["quadratmeter", "cm²", "rechteck", "dreieck"], "volumen": ["liter", "cm³", "körper", "zylinder"], "funktion": ["graph", "gerade", "parabel"], "proportional": ["verhältnis", "faktor", "steigung"], "dreieck": ["hypotenuse", "kathete", "rechtwinklig"], "würfel": ["quader", "körper", "kanten"], "wahrscheinlichkeit": ["zufall", "würfel", "karten", "stochastik"] }; // ------------------------------------------------------------- // Levenshtein-Distanz (Fuzzy-Matching für Tippfehler) // ------------------------------------------------------------- function levenshtein(a, b) { const matrix = []; for (let i = 0; i <= b.length; i++) { matrix[i] = [i]; } for (let j = 0; j <= a.length; j++) { matrix[0][j] = j; } for (let i = 1; i <= b.length; i++) { for (let j = 1; j <= a.length; j++) { matrix[i][j] = Math.min( matrix[i-1][j] + 1, matrix[i][j-1] + 1, matrix[i-1][j-1] + (a[j-1] === b[i-1] ? 0 : 1) ); } } return matrix[b.length][a.length]; } // ------------------------------------------------------------- // Automatische Keyword-Erweiterung // ------------------------------------------------------------- function generateKeywordVariants(baseWords) { const variants = new Set(); baseWords.forEach(word => { const w = word.toLowerCase(); // Original variants.add(w); // Pluralformen variants.add(w + "e"); variants.add(w + "en"); variants.add(w + "er"); variants.add(w + "s"); // Bindestrich-Varianten variants.add(w + "-"); variants.add(w + "-gesetz"); variants.add(w + "-formel"); // Kurzformen if (w.length > 4) { variants.add(w.slice(0, 3)); variants.add(w.slice(0, w.length - 2)); } // Tippfehler-Varianten (1 Buchstabe vertauscht) for (let i = 0; i < w.length - 1; i++) { variants.add( w.slice(0, i) + w[i + 1] + w[i] + w.slice(i + 2) ); } // Englische Varianten if (w === "potenz") { variants.add("power"); variants.add("power function"); } if (w === "funktion") { variants.add("function"); variants.add("graph"); } if (w === "winkel") { variants.add("angle"); } if (w === "fläche") { variants.add("area"); } if (w === "volumen") { variants.add("volume"); } }); return Array.from(variants); } // ------------------------------------------------------------- // Haupt-Suchfunktion (Synonyme + Fuzzy-Matching + Ranking) // ------------------------------------------------------------- function search(term) { term = term.toLowerCase().trim(); if (!term) return []; let terms = term.split(/\s+/); // Synonyme ergänzen let expandedTerms = new Set(terms); terms.forEach(t => { if (synonyms[t]) { synonyms[t].forEach(s => expandedTerms.add(s)); } }); let results = []; searchIndex.forEach(entry => { let score = 0; expandedTerms.forEach(t => { entry.keywords.forEach(k => { // exakter Treffer if (k.includes(t)) score += 3; // fuzzy Treffer (Tippfehler, ähnliche Wörter) if (levenshtein(k, t) <= 2) score += 2; }); }); if (score > 0) { results.push({ entry, score }); } }); // Treffer nach Relevanz sortieren results.sort((a, b) => b.score - a.score); return results.map(r => r.entry); }
